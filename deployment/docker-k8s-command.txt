# Real backend: KSS.Service.Project.

# Do NOT use Helper\deployment\build_all.bat — its 19 cd targets point at
# C:\Source Control\KSS, which does not exist, so every chain dies before
# docker runs. Build from this directory.

# Step 1: Login to Docker registry
docker login docker-hosted.sebaoffice.ir

# Step 2: Build the Docker image (pass basePath so assets resolve under /project)
docker build --build-arg NEXT_PUBLIC_BASE_PATH=/project --build-arg NEXT_PUBLIC_APP_NAME=KSS_Client_Web_project -t docker-hosted.sebaoffice.ir/kss-client-web-project:latest .

# Alternative: Build and push in one command (if using buildx)
docker buildx build --push --build-arg NEXT_PUBLIC_BASE_PATH=/project --build-arg NEXT_PUBLIC_APP_NAME=KSS_Client_Web_project -t docker-hosted.sebaoffice.ir/kss-client-web-project:latest .

# NextAuth: this app does not issue sessions, it decrypts the cookie the Shell issued.
# nextauth-secret must be byte-identical to the Shell's. deployment.yaml omits the key on
# purpose — absent + non-optional gives CreateContainerConfigError, so the pod refuses to
# start until step 6 runs. An empty value would instead start the pod and 500 every route
# while the probes keep reporting Ready.
#   kubectl patch secret kss-client-web-project-secrets -n kss-client-web-project -p "{\"data\":{\"nextauth-secret\":\"$(kubectl get secret kss-client-web-secrets -n kss-client-web -o jsonpath='{.data.nextauth-secret}')\"}}"
#   kubectl rollout restart deployment/kss-client-web-project -n kss-client-web-project

# Step 3: Push the image to registry
docker push docker-hosted.sebaoffice.ir/kss-client-web-project:latest

# Step 4: Create secrets (first time only — copy nextauth-secret from the Shell, do not invent one)
#   kubectl create secret generic kss-client-web-project-secrets -n kss-client-web-project \
#     --from-literal=nextauth-secret="$(kubectl get secret kss-client-web-secrets -n kss-client-web -o jsonpath='{.data.nextauth-secret}' | base64 -d)" \
#     --from-literal=storage-endpoint='' --from-literal=storage-region='' \
#     --from-literal=storage-bucket='' --from-literal=storage-access-key-id='' \
#     --from-literal=storage-secret-access-key='' --from-literal=storage-cdn-url='' \
#     --dry-run=client -o yaml | kubectl apply -f -

# Step 5: Apply manifests (Namespace + Deployment + Service + ConfigMap + Secret + PDB + NetworkPolicy)
# EXPECTED: pods will NOT start yet — nextauth-secret is deliberately absent until step 6.
kubectl apply -f deployment.yaml

# Step 6: NEXTAUTH_SECRET — copy it across, do not read it, retype it or generate one
kubectl patch secret kss-client-web-project-secrets -n kss-client-web-project -p "{\"data\":{\"nextauth-secret\":\"$(kubectl get secret kss-client-web-secrets -n kss-client-web -o jsonpath='{.data.nextauth-secret}')\"}}"
# Verify the two print IDENTICAL strings:
#   kubectl get secret kss-client-web-secrets     -n kss-client-web     -o jsonpath='{.data.nextauth-secret}'
#   kubectl get secret kss-client-web-project-secrets -n kss-client-web-project -o jsonpath='{.data.nextauth-secret}'

# Step 7: Restart deployment so pods pick up ConfigMap/Secret changes
kubectl rollout restart deployment/kss-client-web-project -n kss-client-web-project
kubectl rollout status deployment/kss-client-web-project -n kss-client-web-project

# Step 8: Point the Shell at this zone (nothing reaches these pods without it)
# ⚠ The Shell NO LONGER has its own /project copy — deleted 2026-08-27. So unlike mpf there is
# no fallback: without this entry /project/* is 404, and if the zone dies it is 500, never the
# old page. Rollback means restoring the Shell's copy from source, not removing a key.
#   ZONES={"project":"http://kss-client-web-project-service.kss-client-web-project.svc.cluster.local"}
#   kubectl rollout restart deployment/kss-client-web -n kss-client-web

# Step 9: Verify the zone hop (only /project/api/health can prove it — the Shell has no such route)
#   200 = the Shell is proxying here.  404 = ZONES is not in effect.
curl -s -o /dev/null -w "%{http_code}\n" https://erp.seba.ir/project/api/health

# Create docker-registry secret (first time only)
# Secrets are namespace-scoped — the one in kss-client-web does NOT carry over.
kubectl create secret docker-registry docker-secret \
  --docker-server=docker-hosted.sebaoffice.ir \
  --docker-username=YOUR_USERNAME \
  --docker-password=YOUR_PASSWORD \
  --namespace kss-client-web-project

# Useful diagnostic commands
kubectl get pods          -n kss-client-web-project
kubectl get secrets       -n kss-client-web-project
kubectl get configmaps    -n kss-client-web-project
kubectl get svc           -n kss-client-web-project
kubectl get networkpolicy -n kss-client-web-project
kubectl get pdb           -n kss-client-web-project

# This zone has no public entrypoint, so there is nothing here to check for
# ingress, certificate, pvc or hpa — those do not exist in this namespace by
# design, not by omission.

# Check pod logs
kubectl logs -n kss-client-web-project -l app=kss-client-web-project --tail=50

# Check pod env vars (confirm NEXTAUTH_SECRET is populated and basePath is /project)
kubectl exec -it $(kubectl get pods -n kss-client-web-project -l app=kss-client-web-project -o jsonpath='{.items[0].metadata.name}') -n kss-client-web-project -- env | sort

# Reach the zone from inside the cluster, bypassing the Shell
kubectl run -n kss-client-web-project tmp-curl --rm -it --image=curlimages/curl --restart=Never -- curl -s -o /dev/null -w "%{http_code}\n" http://kss-client-web-project-service.kss-client-web-project.svc.cluster.local/project/api/health

# Delete resources if needed
kubectl delete networkpolicy kss-client-web-project-network-policy -n kss-client-web-project
kubectl delete secrets docker-secret -n kss-client-web-project
